  function esc(t) {

  function md(t) {
    var x = esc(t);
    x = x.replace(/\*\*([^*]+)\*\*/g, "<b>$1</b>");
    x = x.replace(/\[([^\]]+)\]\((https?:[^\s)]+)\)/g, '<a href="$2" target="_blank" rel="noopener">$1</a>');
    x = x.replace(/(^|[\s>])(https?:\/\/[^\s<]+)/g, '$1<a href="$2" target="_blank" rel="noopener">$2</a>');
    x = x.replace(/^\s*[-*] (.*)$/gm, "• $1");
    x = x.replace(/\n/g, "<br>");
    return x;
  }

  function textColorFor(hex, fallback) {
    var m = /^#?([0-9a-f]{6})$/i.exec(hex || "");
    if (!m) return fallback || "#fff";
    var n = parseInt(m[1], 16);
    var l = (0.299 * ((n >> 16) & 255) + 0.587 * ((n >> 8) & 255) + 0.114 * (n & 255)) / 255;
    return l > 0.6 ? "#1a1a1a" : "#fff";
  }

  function playPing() {
    try {
      var ac = new (window.AudioContext || window.webkitAudioContext)();
      if (ac.state === "suspended") return; // el navegador exige interacción previa
      var play = function (freq, at) {
        var o = ac.createOscillator();
        var g = ac.createGain();
        o.type = "sine";
        o.frequency.value = freq;
        g.gain.setValueAtTime(0.0001, ac.currentTime + at);
        g.gain.exponentialRampToValueAtTime(0.04, ac.currentTime + at + 0.02);
        g.gain.exponentialRampToValueAtTime(0.0001, ac.currentTime + at + 0.25);
        o.connect(g).connect(ac.destination);
        o.start(ac.currentTime + at);
        o.stop(ac.currentTime + at + 0.3);
      };
      play(830, 0);
      play(1108, 0.12);
    } catch (e) {}
  }

      '<div><div class="cb-hname">' + (TH.wordmark_url ? '<img class="cb-hwordmark" src="' + esc(TH.wordmark_url) + '" alt="' + esc(cfg.name) + '">' : esc(cfg.name)) + '</div>' +
      '<div class="cb-hsub">' + esc(TXT.subtitle) + '</div></div>' +
      '<button class="cb-close" aria-label="Cerrar chat">&times;</button></div>' +
      '<div class="cb-log" id="cb-log"></div>' +
      '<div class="cb-sug" id="cb-sug"></div>' +
      '<div class="cb-foot"><input id="cb-in" placeholder="' + esc(TXT.placeholder) + '" ' +
      'aria-label="Tu pregunta" autocomplete="off"><button id="cb-send" aria-label="Enviar">' +
      (SEND_ICONS[TH.icon_send]
        ? '<svg viewBox="0 0 24 24">' + SEND_ICONS[TH.icon_send] + "</svg>"
        : esc(TXT.send)) +
      "</button></div>" + brand;

    panel.querySelector(".cb-close").onclick = toggle;
    panel.querySelector("#cb-send").onclick = send;
    var inp = panel.querySelector("#cb-in");
    inp.onkeydown = function (e) {
      if (e.key === "Enter") send();
    };
    inp.onfocus = function () { setTimeout(fitMobile, 250); };
    inp.onblur = function () { setTimeout(fitMobile, 250); };

    add("bot", cfg.welcome_message);

    if ((TH.qualify_opts || []).length && !QUAL) {
      showQualify();
    } else {
      renderSug();
    }
  }

  function renderSug() {
    var sug = panel.querySelector("#cb-sug");
    if (!sug || sug.childNodes.length) return;
    (cfg.suggested_questions || []).forEach(function (q) {
      var b = document.createElement("button");
      b.textContent = q;
      b.onclick = function () {
        panel.querySelector("#cb-in").value = q;
        send();
      };
      sug.appendChild(b);
    });
  }

  function showQualify() {
    add("bot", TH.qualify_q || "Para ayudarte mejor, cuéntame quién eres:");
    var log = panel.querySelector("#cb-log");
    var box = document.createElement("div");
    box.className = "cb-qual";
    (TH.qualify_opts || []).slice(0, 4).forEach(function (opt) {
      var b = document.createElement("button");
      b.textContent = opt;
      b.onclick = function () {
        QUAL = opt;
        sessionStorage.setItem("cb_qual", opt);
        box.remove();
        panel.querySelector("#cb-in").value = opt;
        send(); // send() limpia #cb-sug, así que renderSug() aquí no tenía efecto
      };
      box.appendChild(b);
    });
    log.appendChild(box);
    scrollLog();
  }

  function kindFromQual(k) {
    var s = (k || QUAL || "").toLowerCase();
    if (s.indexOf("exposit") >= 0) return "expositor";
    if (s.indexOf("visit") >= 0) return "visitante";
    if (s.indexOf("prensa") >= 0 || s.indexOf("medio") >= 0) return "prensa";
    return k || "general";
  }

  function showLeadForm(kind) {
    if (sessionStorage.getItem("cb_lead_done") || panel.querySelector(".cb-form")) return;
    var log = panel.querySelector("#cb-log");
    var f = document.createElement("div");
    f.className = "cb-form";
    function field(ph, type, req) {
      var i = document.createElement(type === "area" ? "textarea" : "input");
      if (type !== "area") i.type = type;
      i.placeholder = ph + (req ? " *" : "");
      if (type === "area") i.rows = 2;
      f.appendChild(i);
      return i;
    }
    var fn = field("Tu nombre", "text", true);
    var fe = field("Tu email", "email", true);
    var fp = field("Teléfono (opcional)", "tel");
    var fc = field("Empresa (opcional)", "text");
    var fm = field("¿Qué necesitas? (opcional)", "area");
    var err = document.createElement("div");
    err.className = "cb-ferr";
    f.appendChild(err);
    var go = document.createElement("button");
    go.className = "cb-fsend";
    go.textContent = "Enviar mis datos";
    f.appendChild(go);
    var skip = document.createElement("button");
    skip.className = "cb-fskip";
    skip.textContent = "Prefiero seguir escribiendo";
    skip.onclick = function () { f.remove(); };
    f.appendChild(skip);
    go.onclick = function () {
      var nm = fn.value.trim();
      var em = fe.value.trim();
      if (!nm || em.indexOf("@") < 1) {
        err.textContent = "Necesitamos al menos tu nombre y un email válido.";
        err.style.display = "block";
        return;
      }
      err.style.display = "none";
      go.disabled = true;
      go.textContent = "Enviando…";
      fetch(API + "/api/lead", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          key: KEY,
          session_id: sid,
          kind: kindFromQual(kind),
          name: nm,
          email: em,
          phone: fp.value.trim(),
          company: fc.value.trim(),
          message: fm.value.trim(),
        }),
      })
        .then(function (r) { return r.json(); })
        .then(function (d) {
          if (d.error) throw new Error(d.error);
          sessionStorage.setItem("cb_lead_done", "1");
          f.remove();
          add("bot", "¡Gracias, " + nm.split(" ")[0] + "! Hemos recibido tus datos y el equipo te contactará muy pronto. ¿Puedo ayudarte con algo más?");
          history.push({ role: "user", content: "(He enviado mis datos de contacto en el formulario: " + nm + ", " + em + ")" });
          history.push({ role: "assistant", content: "Datos recibidos. El equipo le contactará pronto." });
        })
        .catch(function (e) {
          go.disabled = false;
          go.textContent = "Enviar mis datos";
          err.textContent = (e && e.message) || "No se ha podido enviar. Prueba otra vez.";
          err.style.display = "block";
        });
    };
    log.appendChild(f);
    scrollLog();
    fn.focus();
  }

  function scrollLog() {
    var log = panel.querySelector("#cb-log");
    if (log) log.scrollTop = log.scrollHeight;
  }

  function typeInto(d, text) {
    var i = 0;
    (function step() {
      i += 3 + Math.floor(Math.random() * 4);
      if (i >= text.length) {
        d.classList.add("md");
        d.innerHTML = md(text);
        scrollLog();
        return;
      }
      d.textContent = text.slice(0, i);
      scrollLog();
      setTimeout(step, 24);
    })();
  }

  function add(who, text, sources, animate) {
    var log = panel.querySelector("#cb-log");
    var d = document.createElement("div");
    d.className = "cb-msg " + (who === "me" ? "me" : "bot");
    if (who === "me") {
      d.textContent = text;
    } else if (animate && !reduced) {
      typeInto(d, text);
    } else {
      d.classList.add("md");
      d.innerHTML = md(text);
    }
    log.appendChild(d);

    if (sources && sources.length) {
      var s2 = document.createElement("div");
      s2.className = "cb-src";
      s2.innerHTML = "Fuente: " + sources.map(function (u) {
        return '<a href="' + esc(u) + '" target="_blank" rel="noopener">' +
          esc(u.replace(/^https?:\/\/[^/]+\//, "").slice(0, 34) || "web") + "</a>";
      }).join(" · ");
      log.appendChild(s2);
    }
    scrollLog();
    return d;
  }

  function send() {
    var input = panel.querySelector("#cb-in");
    var text = input.value.trim();
    if (!text || busy) return;

    busy = true;
    input.value = "";
    panel.querySelector("#cb-sug").innerHTML = "";
    panel.querySelector("#cb-send").disabled = true;
    add("me", text);

    var wait = add("bot", "");
    wait.className = "cb-msg bot cb-dots";
    wait.innerHTML = "<span></span><span></span><span></span>";

    fetch(API + "/api/chat", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        key: KEY,
        session_id: sid,
        message: text,
        page_url: location.href,
        history: history,
      }),
    })
      .then(function (r) { return r.json(); })
      .then(function (d) {
        wait.remove();
        if (d.error) throw new Error(d.error);
        add("bot", d.reply, d.sources, true);
        history.push({ role: "user", content: text });
        history.push({ role: "assistant", content: d.reply });
        if (d.lead_form) {
          var delay = reduced ? 0 : Math.min(d.reply.length * 9, 1800);
          setTimeout(function () { showLeadForm(d.lead_form.kind); }, delay);
        }
      })
      .catch(function () {
        wait.remove();
        add("bot", TXT.error);
      })
      .finally(function () {
        busy = false;
        panel.querySelector("#cb-send").disabled = false;
        input.focus();
      });
  }

  function hideTeaser() {
    if (teaser) { teaser.remove(); teaser = null; }
  }

  function showTeaser() {
    if (open || teaser || sessionStorage.getItem("cb_teased")) return;
    teaser = document.createElement("div");
    teaser.className = "cb-tease" +
      (TH.position === "izquierda" ? " cb-left" : "") +
      (isDark() ? " cb-dark" : "");
    teaser.setAttribute("role", "button");
    teaser.innerHTML = '<button class="cb-tx" aria-label="Cerrar aviso">&times;</button>' + esc(TXT.teaser);
    teaser.onclick = function (e) {
      var closeOnly = e.target && e.target.className === "cb-tx";
      hideTeaser();
      if (!closeOnly) openChat();
    };
    document.body.appendChild(teaser);
    sessionStorage.setItem("cb_teased", "1");
    if (TH.sound) playPing();
  }

  function isMobile() {
    return window.innerWidth <= 480;
  }

  // iOS desplaza la zona visible cuando abre el teclado: el panel debe
  // encogerse a su alto Y seguirla (offsetTop), no solo encogerse
  function fitMobile() {
    var vv = window.visualViewport;
    if (!isMobile() || !open || !vv) {
      panel.style.height = "";
      panel.style.top = "";
      panel.style.bottom = "";
      return;
    }
    panel.style.top = Math.round(vv.offsetTop) + "px";
    panel.style.bottom = "auto";
    panel.style.height = Math.round(vv.height) + "px";
    setTimeout(scrollLog, 60);
  }

  if (window.visualViewport) {
    window.visualViewport.addEventListener("resize", fitMobile);
    window.visualViewport.addEventListener("scroll", fitMobile);
  }
  window.addEventListener("resize", fitMobile);

  // bloqueo de la página al estilo iOS: se fija en su posición actual
  // (overflow:hidden a secas no basta en Safari) y se restaura al cerrar
  var savedScroll = 0;
  function lockPage(on) {
    if (!isMobile()) on = false;
    var b = document.body.style;
    if (on) {
      savedScroll = window.scrollY || window.pageYOffset || 0;
      b.position = "fixed";
      b.top = -savedScroll + "px";
      b.left = "0";
      b.right = "0";
      b.width = "100%";
      document.documentElement.style.overflow = "hidden";
    } else {
      b.position = "";
      b.top = "";
      b.left = "";
      b.right = "";
      b.width = "";
      document.documentElement.style.overflow = "";
      if (savedScroll) window.scrollTo(0, savedScroll);
      savedScroll = 0;
    }
  }

  function toggle() {
    open = !open;
    panel.classList.toggle("on", open);
    lockPage(open);
    fitMobile();
    if (open) {
      hideTeaser();
      if (!isMobile()) panel.querySelector("#cb-in").focus();
      scrollLog();
    }
  }

  function openChat() {
    if (!panel.innerHTML) render();
    if (!open) toggle();
  }

  btn.onclick = function () {
    hideTeaser();
    if (!panel.innerHTML) render();
    toggle();
  };

  fetch(API + "/api/config?key=" + encodeURIComponent(KEY))
    .then(function (r) { return r.json(); })
    .then(function (d) {
      if (d.error) return;
      cfg = d;
      applyTheme();
      if (AUTO > 0) {
        setTimeout(openChat, AUTO);
      } else if (!TH || TH.teaser !== false) {
        setTimeout(showTeaser, Math.max(1, (TH && TH.teaser_delay) || 4) * 1000);
      }
    })
    .catch(function () {});

  applyTheme();
})();
