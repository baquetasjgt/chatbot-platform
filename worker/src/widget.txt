  function esc(t) {
    var d = document.createElement("div");
    d.textContent = t;
    // textContent no escapa comillas; esc() también se usa dentro de atributos
    return d.innerHTML.replace(/"/g, "&quot;").replace(/'/g, "&#39;");
  }

  function md(t) {
    var x = esc(t);
    x = x.replace(/\*\*([^*]+)\*\*/g, "<b>$1</b>");
    x = x.replace(/\[([^\]]+)\]\((https?:[^\s)]+)\)/g, '<a href="$2" target="_blank" rel="noopener">$1</a>');
    x = x.replace(/(^|[\s>])(https?:\/\/[^\s<]+)/g, '$1<a href="$2" target="_blank" rel="noopener">$2</a>');
    x = x.replace(/^\s*[-*] (.*)$/gm, "• $1");
    x = x.replace(/\n/g, "<br>");
    return x;
  }

  function textColorFor(hex, fallback) {
    var m = /^#?([0-9a-f]{6})$/i.exec(hex || "");
    if (!m) return fallback || "#fff";
    var n = parseInt(m[1], 16);
    var l = (0.299 * ((n >> 16) & 255) + 0.587 * ((n >> 8) & 255) + 0.114 * (n & 255)) / 255;
    return l > 0.6 ? "#1a1a1a" : "#fff";
  }

    return "'" + f + "',system-ui,sans-serif";
  }

  function playPing() {
    try {
      var ac = new (window.AudioContext || window.webkitAudioContext)();
      if (ac.state === "suspended") return; // el navegador exige interacción previa
      var play = function (freq, at) {
        var o = ac.createOscillator();
        var g = ac.createGain();
        o.type = "sine";
        o.frequency.value = freq;
        g.gain.setValueAtTime(0.0001, ac.currentTime + at);
        g.gain.exponentialRampToValueAtTime(0.04, ac.currentTime + at + 0.02);
        g.gain.exponentialRampToValueAtTime(0.0001, ac.currentTime + at + 0.25);
        o.connect(g).connect(ac.destination);
        o.start(ac.currentTime + at);
        o.stop(ac.currentTime + at + 0.3);
      };
      play(830, 0);
      play(1108, 0.12);
    } catch (e) {}
  }

  function typeInto(d, text) {
    var i = 0;
    (function step() {
      i += 3 + Math.floor(Math.random() * 4);
      if (i >= text.length) {
        d.classList.add("md");
        d.innerHTML = md(text);
        scrollLog();
        return;
      }
      d.textContent = text.slice(0, i);
      scrollLog();
      setTimeout(step, 24);
    })();
  }

  function add(who, text, sources, animate) {
    var log = panel.querySelector("#cb-log");
    var d = document.createElement("div");
    d.className = "cb-msg " + (who === "me" ? "me" : "bot");
    if (TH.timestamps) {
      var now = new Date();
      d.setAttribute("data-t", ("0" + now.getHours()).slice(-2) + ":" + ("0" + now.getMinutes()).slice(-2));
    }
    if (who === "me") {
      d.textContent = text;
    } else if (animate && !reduced) {
      typeInto(d, text);
    } else {
      d.classList.add("md");
      d.innerHTML = md(text);
    }
    log.appendChild(d);

    if (sources && sources.length) {
      var s2 = document.createElement("div");
      s2.className = "cb-src";
      s2.innerHTML = "Fuente: " + sources.map(function (u) {
        return '<a href="' + esc(u) + '" target="_blank" rel="noopener">' +
          esc(u.replace(/^https?:\/\/[^/]+\//, "").slice(0, 34) || "web") + "</a>";
      }).join(" · ");
      log.appendChild(s2);
    }
    scrollLog();
    return d;
  }

  function send() {
    var input = panel.querySelector("#cb-in");
    var text = input.value.trim();
    if (!text || busy) return;

    busy = true;
    input.value = "";
    panel.querySelector("#cb-sug").innerHTML = "";
    panel.querySelector("#cb-send").disabled = true;
    add("me", text);

    var wait = add("bot", "");
    wait.className = "cb-msg bot cb-dots";
    wait.innerHTML = "<span></span><span></span><span></span>";

    fetch(API + "/api/chat", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        key: KEY,
        session_id: sid,
        message: text,
        page_url: location.href,
        history: history,
      }),
    })
      .then(function (r) { return r.json(); })
      .then(function (d) {
        wait.remove();
        if (d.error) throw new Error(d.error);
        add("bot", d.reply, d.sources, true);
        history.push({ role: "user", content: text });
        history.push({ role: "assistant", content: d.reply });
        if (d.lead_form) {
          var delay = reduced ? 0 : Math.min(d.reply.length * 9, 1800);
          setTimeout(function () { showLeadForm(d.lead_form.kind); }, delay);
        }
      })
      .catch(function () {
        wait.remove();
        add("bot", TXT.error);
      })
      .finally(function () {
        busy = false;
        panel.querySelector("#cb-send").disabled = false;
        input.focus();
      });
  }

  // iOS desplaza la zona visible cuando abre el teclado: el panel debe
  // encogerse a su alto Y seguirla (offsetTop), no solo encogerse
  function fitMobile() {
    var vv = window.visualViewport;
    if (!isMobile() || !open || !vv) {
      panel.style.height = "";
      panel.style.top = "";
      panel.style.bottom = "";
      return;
    }
    panel.style.top = Math.round(vv.offsetTop) + "px";
    panel.style.bottom = "auto";
    panel.style.height = Math.round(vv.height) + "px";
    setTimeout(scrollLog, 60);
  }

  if (window.visualViewport) {
    window.visualViewport.addEventListener("resize", fitMobile);
    window.visualViewport.addEventListener("scroll", fitMobile);
  }
  window.addEventListener("resize", fitMobile);

  // bloqueo de la página al estilo iOS: se fija en su posición actual
  // (overflow:hidden a secas no basta en Safari) y se restaura al cerrar
  var savedScroll = 0;
  function lockPage(on) {
    if (!isMobile()) on = false;
    var b = document.body.style;
    if (on) {
      savedScroll = window.scrollY || window.pageYOffset || 0;
      b.position = "fixed";
      b.top = -savedScroll + "px";
      b.left = "0";
      b.right = "0";
      b.width = "100%";
      document.documentElement.style.overflow = "hidden";
    } else {
      b.position = "";
      b.top = "";
      b.left = "";
      b.right = "";
      b.width = "";
      document.documentElement.style.overflow = "";
      if (savedScroll) window.scrollTo(0, savedScroll);
      savedScroll = 0;
    }
  }

  function toggle() {
    open = !open;
    panel.classList.toggle("on", open);
    lockPage(open);
    fitMobile();
    if (open) {
      hideTeaser();
      if (!isMobile()) panel.querySelector("#cb-in").focus();
      scrollLog();
    }
  }

  function openChat() {
    if (!panel.innerHTML) render();
    if (!open) toggle();
  }

  btn.onclick = function () {
    hideTeaser();
    if (!panel.innerHTML) render();
    toggle();
  };

  fetch(API + "/api/config?key=" + encodeURIComponent(KEY))
    .then(function (r) { return r.json(); })
    .then(function (d) {
      if (d.error) return;
      cfg = d;
      applyTheme();
      if (AUTO > 0) {
        setTimeout(openChat, AUTO);
      } else if (!TH || TH.teaser !== false) {
        setTimeout(showTeaser, Math.max(1, (TH && TH.teaser_delay) || 4) * 1000);
      }
    })
    .catch(function () {});

  applyTheme();
})();
