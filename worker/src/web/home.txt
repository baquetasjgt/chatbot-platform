  function C(v) { return Math.max(0, Math.min(1, v)); }
  function seg(p, a, b) { return C((p - a) / (b - a)); }
  function S(el, o, tr) { if (el) { el.style.opacity = o; if (tr !== undefined) el.style.transform = tr; } }
  function updateFx() {
    var vh = window.innerHeight;
    var doc = document.documentElement;
    $("scrollbar").style.transform = "scaleX(" + C(doc.scrollTop / (doc.scrollHeight - vh)) + ")";
    document.querySelectorAll("[data-scene]").forEach(function (sc) {
      var r = sc.getBoundingClientRect();
      if (r.bottom < -vh || r.top > vh * 2) return;
      var pinned = sc.offsetHeight > vh * 1.5;
      var p = pinned ? C(-r.top / (sc.offsetHeight - vh)) : C((vh - r.top) / (vh * 0.7));
      var F = function (n) { return sc.querySelector('[data-fx="' + n + '"]'); };
      var k = sc.dataset.scene;
      if (k === "hero") {
        ["q1", "q2", "q3"].forEach(function (n, i) {
          var el = F(n); if (!el) return;
          var tIn = seg(p, 0.04 + 0.1 * i, 0.18 + 0.1 * i), tCv = seg(p, 0.55, 0.75);
          S(el, Math.min(tIn, 1 - 0.7 * tCv), "translate(" + (+el.dataset.cx * tCv) + "px," + (+el.dataset.cy * tCv + 18 * (1 - tIn)) + "px) scale(" + (1 - 0.25 * tCv) + ")");
        });
        var tb = seg(p, 0.34, 0.5); S(F("bot"), tb, "translate(-50%,-50%) scale(" + (0.4 + 0.6 * tb) + ")");
        S(F("lines"), seg(p, 0.48, 0.6));
        var tl = seg(p, 0.6, 0.76); S(F("lead"), tl, "translateX(" + (26 * (1 - tl)) + "px)");
        var tm = seg(p, 0.78, 0.94); S(F("msg2"), tm, "translateY(" + (14 * (1 - tm)) + "px)");
        var bw = F("bigword"); if (bw) bw.style.transform = "translateX(" + (-p * 90) + "px)";
      } else if (k === "lead") {
        var w = [[0, 0.28], [0.24, 0.52], [0.48, 0.76], [0.72, 1]];
        for (var i = 0; i < 4; i++) {
          var el = F("st" + (i + 1)); if (!el) continue;
          var tIn = seg(p, w[i][0], w[i][0] + 0.08);
          var tOut = i < 3 ? 1 - seg(p, w[i][1] - 0.07, w[i][1]) : 1;
          S(el, Math.min(tIn, tOut), "translateY(" + (20 * (1 - tIn) - 14 * (1 - tOut)) + "px) scale(" + (0.97 + 0.03 * Math.min(tIn, tOut)) + ")");
        }
        var rail = F("rail"); if (rail) rail.style.transform = "scaleX(" + p + ")";
        var lp = F("leadpath"); if (lp) lp.style.strokeDashoffset = 1 - p;
      } else if (k === "canales") {
        sc.querySelectorAll('[data-fx="ch"]').forEach(function (el, i) {
          var t = seg(p, 0.08 + 0.2 * i, 0.3 + 0.2 * i); S(el, t, "translateY(" + (34 * (1 - t)) + "px)");
        });
        var tc = seg(p, 0.72, 0.9); S(F("conv"), tc, "translateY(" + (16 * (1 - tc)) + "px)");
      } else if (k === "feria") {
        if (window.innerWidth <= 760) return;
        var tr = F("htrack"), cl = sc.querySelector(".feria-clip");
        if (tr && cl) {
          var max = Math.max(0, tr.scrollWidth - cl.clientWidth);
          tr.style.transform = "translateX(" + (-seg(p, 0.05, 0.95) * max) + "px)";
        }
      } else if (k === "bars") {
        sc.querySelectorAll('[data-fx="bar"]').forEach(function (el, i) {
          el.style.transform = "scaleY(" + Math.min(1, 0.15 + 0.85 * seg(p, 0.05 + i * 0.03, 0.5 + i * 0.03)) + ")";
        });
        sc.querySelectorAll("[data-cnt]").forEach(function (el) {
          var v = Math.round(+el.dataset.cnt * seg(p, 0.05, 0.6));
